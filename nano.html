# --- 2. POSITIES AAN DE HEMEL (Rechte Klimming & Declinatie - Epoch 2041) -----
eq_to_horiz <- function(ra_deg, dec_deg, lst_deg, lat_r) {
  ha_deg <- ((lst_deg - ra_deg + 180) %% 360) - 180
  ha_rad <- ha_deg * pi / 180
  dec_rad <- dec_deg * pi / 180
  sin_alt <- sin(dec_rad) * sin(lat_r) + cos(dec_rad) * cos(lat_r) * cos(ha_rad)
  alt_rad <- asin(pmax(pmin(sin_alt, 1), -1))
  y <- -sin(ha_rad) * cos(dec_rad)
  x <- sin(dec_rad) * cos(lat_r) - cos(dec_rad) * sin(lat_r) * cos(ha_rad)
  az_deg <- (atan2(y, x) * 180 / pi) %% 360
  c(Alt_deg = round(alt_rad * 180 / pi, 2), Az_deg = round(az_deg, 2), Uurhoek_deg = round(ha_deg, 2))
}
